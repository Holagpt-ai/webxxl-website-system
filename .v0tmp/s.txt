export type GscKeywordListItem = {
  query: string
  page: string | null
  clicks: number | null
  impressions: number | null
  ctr: number | null
  averagePosition: number | null
  averagePositionLabel: 'GSC average position (not fixed rank)'
  date: string
  dataState: string
  observedAt: string
}

type DedupedRow = GscKeywordListItem & {
  sortDateMs: number
  sortQuery: string
  sortPage: string
  sortId: string
}

function clampTake(take?: number): number {
  return Math.min(Math.max(Number(take ?? SEO_LIST_DEFAULT_TAKE) || SEO_LIST_DEFAULT_TAKE, 1), SEO_LIST_MAX_TAKE)
}

function toIsoDate(d: Date): string {
  return d.toISOString().slice(0, 10)
}

export function dedupeKeywordRowsForList(rows: PersistedKeywordRow[]): DedupedRow[] {
  const byMetric = new Map<string, DedupedRow>()
  for (const row of rows) {
    const metricKey = `${row.query}|${row.page ?? ''}|${toIsoDate(row.date)}`
    const existing = byMetric.get(metricKey)
    const candidate: DedupedRow = {
      query: row.query,
      page: row.page,
      clicks: row.clicks,
      impressions: row.impressions,
      ctr: row.ctr,
      averagePosition: row.averagePosition,
      averagePositionLabel: 'GSC average position (not fixed rank)',
      date: toIsoDate(row.date),
      dataState: row.dataState,
      observedAt: row.observedAt.toISOString(),
      sortDateMs: row.date.getTime(),
      sortQuery: row.query,
      sortPage: row.page ?? '',
      sortId: row.idempotencyKey,
    }
    if (!existing || row.observedAt > new Date(existing.observedAt)) {
      byMetric.set(metricKey, candidate)
    }
  }
  return [...byMetric.values()].sort(compareKeywordSortKeys)
}

/** date DESC, query ASC, page ASC, idempotencyKey ASC */
export function compareKeywordSortKeys(a: DedupedRow, b: DedupedRow): number {
  if (b.sortDateMs !== a.sortDateMs) return b.sortDateMs - a.sortDateMs
  const q = a.sortQuery.localeCompare(b.sortQuery)
  if (q !== 0) return q
  const p = a.sortPage.localeCompare(b.sortPage)
  if (p !== 0) return p
  return a.sortId.localeCompare(b.sortId)
}

function keywordCursorFromRow(row: DedupedRow): SeoListCursorPayload {
  return { p: row.sortDateMs, q: row.sortQuery, pg: row.sortPage, i: row.sortId }
}

function cursorSortAnchor(cursor: SeoListCursorPayload): DedupedRow {
  return {
    query: cursor.q ?? '',
    page: cursor.pg ? cursor.pg : null,
    clicks: null,
    impressions: null,
    ctr: null,
    averagePosition: null,
    averagePositionLabel: 'GSC average position (not fixed rank)',
    date: '',
    dataState: 'READY',
    observedAt: '',
    sortDateMs: cursor.p,
    sortQuery: cursor.q ?? '',
    sortPage: cursor.pg ?? '',
    sortId: cursor.i,
  }
}

/** Rows strictly after cursor in deterministic sort order. */
export function keywordRowsAfterCursor(sorted: DedupedRow[], cursor: SeoListCursorPayload): DedupedRow[] {
  const anchor = cursorSortAnchor(cursor)
  return sorted.filter((row) => compareKeywordSortKeys(anchor, row) < 0)
}

export async function listGscKeywordsForSite(input: {
  customerAccountId: string
  managedSiteId: string
  take?: number
  cursor?: string
  search?: string
}): Promise<
  SeoResult<{
    gsc: GscPerformanceReadModel
    keywords: GscKeywordListItem[]
    nextCursor?: string
    limitations: string[]
  }>
> {
  const bundle = await loadGscStoredSiteBundle({
    customerAccountId: input.customerAccountId,
    managedSiteId: input.managedSiteId,
  })
  if (!bundle.ok) return bundle

  const take = clampTake(input.take)
  let sorted = dedupeKeywordRowsForList(bundle.value.rows)
  const q = input.search?.trim().toLowerCase()
  if (q) {
    sorted = sorted.filter(
      (row) => row.query.toLowerCase().includes(q) || (row.page ?? '').toLowerCase().includes(q),
    )
  }

  if (input.cursor) {
    const decoded = decodeSeoListCursor(input.cursor)
    if (!decoded.ok) return decoded
    sorted = keywordRowsAfterCursor(sorted, decoded.value)
  }

  const page = sorted.slice(0, take)
  const keywords: GscKeywordListItem[] = page.map(
    ({ sortDateMs: _d, sortQuery: _q, sortPage: _p, sortId: _i, ...rest }) => rest,
  )

  let nextCursor: string | undefined
  if (sorted.length > take) {
    const last = page[page.length - 1]!
    nextCursor = encodeSeoListCursor(keywordCursorFromRow(last))
  }

  return {
    ok: true,
    value: {
      gsc: bundle.value.performance,
      keywords,
      nextCursor,
      limitations: bundle.value.limitations,
    },
  }
}

function parseOptionalEnum<T extends string>(raw: string | undefined, allowed: readonly T[], label: string): SeoResult<T | undefined> {
  if (!raw) return { ok: true, value: undefined }
  if (!allowed.includes(raw as T)) {
    return { ok: false, code: 'VALIDATION', message: `Invalid ${label} filter` }
  }
  return { ok: true, value: raw as T }
}

const OPPORTUNITY_STATES = [
  'OBSERVED_FINDING',
  'RECOMMENDED_ACTION',
  'DRAFT_PREPARED',
  'AWAITING_APPROVAL',
  'APPROVED',
  'PUBLISHED',
  'VERIFIED_OUTCOME',
] as const
const LEVELS = ['LOW', 'MEDIUM', 'HIGH'] as const

function cursorWhere(
  base: Prisma.SeoOpportunityWhereInput,
  cursor: { p: number; i: string },
): Prisma.SeoOpportunityWhereInput {
  return {
    AND: [
      base,
      {
        OR: [{ priorityScore: { lt: cursor.p } }, { priorityScore: cursor.p, id: { gt: cursor.i } }],
      },
    ],
  }
}

export async function listSeoOpportunitiesForSite(input: {
  customerAccountId: string
  managedSiteId: string
  take?: number
  cursor?: string
  state?: string
  severity?: string
  confidence?: string
}): Promise<SeoResult<{ opportunities: Awaited<ReturnType<typeof prisma.seoOpportunity.findMany>>; nextCursor?: string }>> {
  const site = await requireManagedSiteInAccount({
    siteId: input.managedSiteId,
    customerAccountId: input.customerAccountId,
  })
  if (!site.ok) return site

  const state = parseOptionalEnum(input.state, OPPORTUNITY_STATES, 'state')
  if (!state.ok) return state
  const severity = parseOptionalEnum(input.severity, LEVELS, 'severity')
  if (!severity.ok) return severity
  const confidence = parseOptionalEnum(input.confidence, LEVELS, 'confidence')
  if (!confidence.ok) return confidence

  const take = clampTake(input.take)
  let where: Prisma.SeoOpportunityWhereInput = {
    managedSiteId: site.value.id,
    customerAccountId: input.customerAccountId,
    ...(state.value ? { state: state.value } : {}),
    ...(severity.value ? { severity: severity.value } : {}),
    ...(confidence.value ? { confidence: confidence.value } : {}),
  }

  if (input.cursor) {
    const decoded = decodeSeoListCursor(input.cursor)
    if (!decoded.ok) return decoded
    where = cursorWhere(where, decoded.value)
  }

  const opportunities = await prisma.seoOpportunity.findMany({
    where,
    orderBy: [{ priorityScore: 'desc' }, { id: 'asc' }],
    take,
  })

  let nextCursor: string | undefined
  if (opportunities.length === take) {
    const last = opportunities[opportunities.length - 1]!
    nextCursor = encodeSeoListCursor({ p: last.priorityScore, i: last.id })
  }

  return { ok: true, value: { opportunities, nextCursor } }
}

/**
 * Opaque-cursor paging. The API returns only `nextCursor`, so Previous walks back through the
 * cursors already visited. Changing `filterKey` drops the history and returns to the first page.
 */
export function useCursorPages(filterKey: string) {
  const [state, setState] = useState<{ key: string; stack: string[] }>({ key: filterKey, stack: [] })
  const stack = state.key === filterKey ? state.stack : []

  const next = useCallback(
    (cursor: string) => setState((s) => ({ key: filterKey, stack: [...(s.key === filterKey ? s.stack : []), cursor] })),
    [filterKey],
  )
  const previous = useCallback(
    () => setState((s) => ({ key: filterKey, stack: (s.key === filterKey ? s.stack : []).slice(0, -1) })),
    [filterKey],
  )
  const first = useCallback(() => setState({ key: filterKey, stack: [] }), [filterKey])

  return {
    cursor: stack.at(-1),
    page: stack.length + 1,
    hasPrevious: stack.length > 0,
    next,
    previous,
    first,
  }
}

export function useDebouncedValue<T>(value: T, delayMs = 300): T {
  const [debounced, setDebounced] = useState(value)
  useEffect(() => {
    const timer = setTimeout(() => setDebounced(value), delayMs)
    return () => clearTimeout(timer)
  }, [value, delayMs])
  return debounced
}
